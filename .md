```python
from flask import Flask, render_template_string, request, send_file
from yt_dlp import YoutubeDL
from werkzeug.utils import secure_filename
import os
import shutil
import subprocess
import tempfile

app = Flask(__name__)
app.config["MAX_CONTENT_LENGTH"] = 200 * 1024 * 1024  # 200 MB

HTML = """

def find_latest_file(directory):
    files = []
    for name in os.listdir(directory):
        full_path = os.path.join(directory, name)
        if os.path.isfile(full_path):
            files.append(full_path)
    if not files:
        return None
    return max(files, key=os.path.getmtime)


def convert_to_mp3(source_path):
    output_dir = os.path.dirname(source_path)
    base_name = os.path.splitext(os.path.basename(source_path))[0]
    output_path = os.path.join(output_dir, f"{base_name}.mp3")

    cmd = [
        "ffmpeg",
        "-y",
        "-i", source_path,
        "-vn",
        "-ar", "44100",
        "-acodec", "libmp3lame",
        "-q:a", "2",
        output_path
    ]

    result = subprocess.run(cmd, capture_output=True, text=True)
    if result.returncode != 0:
        raise RuntimeError(result.stderr or "FFmpeg conversion failed.")

    return output_path


def download_media_from_url(url):
    temp_dir = tempfile.mkdtemp(prefix="media_ripper_")
    ydl_opts = {
        "format": "bestaudio/best",
        "quiet": True,
        "noplaylist": True,
        "restrictfilenames": False,
        "no_warnings": True,
        "outtmpl": os.path.join(temp_dir, "%(title)s.%(ext)s"),
    }

    try:
        with YoutubeDL(ydl_opts) as ydl:
            ydl.extract_info(url, download=True)

        downloaded_file = find_latest_file(temp_dir)
        if not downloaded_file:
            raise RuntimeError("No media file was downloaded.")

        return downloaded_file, temp_dir
    except Exception:
        shutil.rmtree(temp_dir, ignore_errors=True)
        raise


@app.route("/")
def home():
    return render_template_string(HTML)


@app.route("/health")
def health():
    return "OK", 200


@app.route("/convert", methods=["POST"])
def convert():
    media_url = (request.form.get("url") or "").strip()
    uploaded_file = request.files.get("file")

    if media_url:
        try:
            source_path, temp_dir = download_media_from_url(media_url)
            mp3_path = convert_to_mp3(source_path)

            response = send_file(
                mp3_path,
                mimetype="audio/mpeg",
                as_attachment=True,
                download_name=os.path.basename(mp3_path),
            )
            response.call_on_close(lambda: shutil.rmtree(temp_dir, ignore_errors=True))
            return response
        except Exception as exc:
            return render_template_string(HTML + f"<p class='error'>Error: {exc}</p>"), 400

    if uploaded_file and uploaded_file.filename:
        filename = secure_filename(uploaded_file.filename)
        temp_dir = tempfile.mkdtemp(prefix="media_upload_")
        source_path = os.path.join(temp_dir, filename)
        uploaded_file.save(source_path)

        try:
            mp3_path = convert_to_mp3(source_path)

            response = send_file(
                mp3_path,
                mimetype="audio/mpeg",
                as_attachment=True,
                download_name=os.path.splitext(filename)[0] + ".mp3",
            )
            response.call_on_close(lambda: shutil.rmtree(temp_dir, ignore_errors=True))
            return response
        except Exception as exc:
            shutil.rmtree(temp_dir, ignore_errors=True)
            return render_template_string(HTML + f"<p class='error'>Error: {exc}</p>"), 400

    return render_template_string(HTML + "<p class='error'>Please enter a URL or upload a file.</p>"), 400


if __name__ == "__main__":
    app.run(host="0.0.0.0", port=8000, debug=True)
```

## Important deployment note

This app calls the system `ffmpeg` binary. On Railway, you usually need FFmpeg installed in the environment as well. If your deployment does not already have it, add a build step or install package support before starting the app:

```bash
apt-get update && apt-get install -y ffmpeg
```
